# 命名管道缓存状态同步系统使用指南

## 概述

`named_pipe_cached_sync.lua` 是一个高效的实时状态同步系统，它结合了应用层缓存和内核缓存机制，通过命名管道实现 Rime 输入法状态的实时同步。该系统主要用于将 Rime 的内部状态（如输入模式、候选词数量等）实时传输给外部应用程序。

## 功能特性

- **双层缓存机制**：结合应用层缓存和内核缓存，减少重复数据传输
- **去重优化**：自动检测并过滤重复的状态数据
- **批处理支持**：支持批量写入，提高传输效率
- **JSON 格式**：使用标准 JSON 格式传输状态数据
- **非阻塞设计**：采用非阻塞模式，不影响输入法性能

## 系统架构

```
┌─────────────────┐    ┌──────────────────┐    ┌─────────────────┐
│  Rime Context   │───▶│  Lua 缓存系统    │───▶│  命名管道       │

```

## 在 Rime Lua 插件中的使用

### 1. 基本集成

在您的 Rime 方案配置文件中添加该模块：

```yaml
# 在 schema.yaml 中
engine:
  processors:
    - lua_processor@named_pipe_cached_sync
  
# 或在 translator 中使用
  translators:
    - lua_translator@your_translator
```

### 2. 在 Lua 插件中使用

```lua
-- 在您的 Lua 插件文件中
```

### 3. update_notifier 的详细使用

```lua
local sync_module = require("named_pipe_cached_sync")

function init(env)
    local context = env.engine.context
    
    -- 监听各种状态变化
    env.update_notifier = context.update_notifier:connect(function(ctx)
        -- 每次输入变化时调用
        sync_module.update_state_cached(ctx)
        
        -- 可以添加额外的处理逻辑
        if ctx:is_composing() then
            log.info("正在输入: " .. (ctx.input or ""))
        end
    end)
    
    -- 监听提交事件
    env.commit_notifier = context.commit_notifier:connect(function(ctx)
        sync_module.update_state_cached(ctx)
        log.info("提交文本")
    end)
    
    -- 监听按键事件
    env.key_notifier = context.unhandled_key_notifier:connect(function(ctx, key)
        sync_module.update_state_cached(ctx)
        log.info("按键: " .. tostring(key))
    end)
    
    -- 监听选项变化
    env.option_notifier = context.option_update_notifier:connect(function(ctx, option)
        sync_module.update_state_cached(ctx)
        log.info("选项变化: " .. tostring(option))
    end)
end
```

## 状态数据格式

系统传输的 JSON 数据格式如下：

```json
{
  "is_composing": true,
  "input_mode": "chinese",
  "input_text": "nihao",
  "candidate_count": 5,
  "schema_name": "wanxiang_pro",
  "timestamp": 1673856000000
}
```

### 字段说明

- `is_composing`: 是否正在输入（布尔值）
- `input_mode`: 输入模式（"chinese" 或 "ascii"）
- `input_text`: 当前输入的文本
- `candidate_count`: 候选词数量
- `schema_name`: 当前使用的输入方案名称
- `timestamp`: 时间戳（毫秒）

## Python 端读取实现

### 1. 基本读取示例

```python
import json
import os
import time
from typing import Dict, Optional

class RimeStateReader:
    def __init__(self, pipe_path: str = "/tmp/rime_state_pipe"):
        self.pipe_path = pipe_path
        self.pipe_fd = None
        
    def connect(self) -> bool:
        """连接到命名管道"""
        try:
            # 以非阻塞模式打开管道
            self.pipe_fd = os.open(self.pipe_path, os.O_RDONLY | os.O_NONBLOCK)
            return True
        except OSError as e:
            print(f"连接管道失败: {e}")
            return False
    
    def read_state(self) -> Optional[Dict]:
        """读取一条状态数据"""
        if not self.pipe_fd:
            return None
            
        try:
            # 读取数据
            data = os.read(self.pipe_fd, 1024)
            if data:
                # 解析 JSON
                json_str = data.decode('utf-8').strip()
                return json.loads(json_str)
        except (OSError, json.JSONDecodeError) as e:
            # 非阻塞读取可能返回空数据，这是正常的
            pass
            
        return None
    
    def close(self):
        """关闭连接"""
        if self.pipe_fd:
            os.close(self.pipe_fd)
            self.pipe_fd = None

# 使用示例
def main():
    reader = RimeStateReader()
    
    if not reader.connect():
        print("无法连接到 Rime 状态管道")
        return
    
    print("开始监听 Rime 状态...")
    
    try:
        while True:
            state = reader.read_state()
            if state:
                print(f"状态更新: {state}")
                
                # 根据状态执行相应操作
                if state.get('is_composing'):
                    print(f"正在输入: {state.get('input_text')}")
                    print(f"候选词数量: {state.get('candidate_count')}")
                else:
                    print("输入完成")
            
            time.sleep(0.01)  # 10ms 轮询间隔
            
    except KeyboardInterrupt:
        print("\n停止监听")
    finally:
        reader.close()

if __name__ == "__main__":
    main()
```

### 2. 异步读取示例

```python
import asyncio
import json
import os
from typing import Dict, Optional, Callable

class AsyncRimeStateReader:
    def __init__(self, pipe_path: str = "/tmp/rime_state_pipe"):
        self.pipe_path = pipe_path
        self.running = False
        self.callbacks = []
        
    def add_callback(self, callback: Callable[[Dict], None]):
        """添加状态变化回调"""
        self.callbacks.append(callback)
    
    async def start_listening(self):
        """开始异步监听"""
        self.running = True
        
        try:
            # 创建管道文件描述符
            pipe_fd = os.open(self.pipe_path, os.O_RDONLY | os.O_NONBLOCK)
            
            while self.running:
                try:
                    data = os.read(pipe_fd, 1024)
                    if data:
                        json_str = data.decode('utf-8').strip()
                        state = json.loads(json_str)
                        
                        # 调用所有回调
                        for callback in self.callbacks:
                            callback(state)
                            
                except (OSError, json.JSONDecodeError):
                    pass
                
                await asyncio.sleep(0.01)
                
        except OSError as e:
            print(f"监听失败: {e}")
        finally:
            if 'pipe_fd' in locals():
                os.close(pipe_fd)
    
    def stop(self):
        """停止监听"""
        self.running = False

# 使用示例
async def main():
    reader = AsyncRimeStateReader()
    
    # 添加状态处理回调
    def on_state_change(state: Dict):
        print(f"异步状态更新: {state}")
        
        if state.get('is_composing'):
            print(f"输入模式: {state.get('input_mode')}")
            print(f"当前输入: {state.get('input_text')}")
        
    reader.add_callback(on_state_change)
    
    # 开始监听
    print("开始异步监听...")
    await reader.start_listening()

if __name__ == "__main__":
    asyncio.run(main())
```

### 3. 高级功能：状态过滤和处理

```python
import json
import os
import time
from typing import Dict, Optional, Set
from dataclasses import dataclass

@dataclass
class RimeState:
    is_composing: bool
    input_mode: str
    input_text: str
    candidate_count: int
    schema_name: str
    timestamp: int
    
    @classmethod
    def from_dict(cls, data: Dict) -> 'RimeState':
        return cls(
            is_composing=data.get('is_composing', False),
            input_mode=data.get('input_mode', 'chinese'),
            input_text=data.get('input_text', ''),
            candidate_count=data.get('candidate_count', 0),
            schema_name=data.get('schema_name', ''),
            timestamp=data.get('timestamp', 0)
        )

class RimeStateProcessor:
    def __init__(self, pipe_path: str = "/tmp/rime_state_pipe"):
        self.pipe_path = pipe_path
        self.pipe_fd = None
        self.last_state: Optional[RimeState] = None
        self.interested_schemas: Set[str] = set()
        
    def add_schema_filter(self, schema_name: str):
        """添加感兴趣的方案过滤器"""
        self.interested_schemas.add(schema_name)
    
    def connect(self) -> bool:
        try:
            self.pipe_fd = os.open(self.pipe_path, os.O_RDONLY | os.O_NONBLOCK)
            return True
        except OSError:
            return False
    
    def process_states(self):
        """处理状态变化"""
        if not self.pipe_fd:
            return
            
        try:
            data = os.read(self.pipe_fd, 1024)
            if data:
                json_str = data.decode('utf-8').strip()
                state_dict = json.loads(json_str)
                current_state = RimeState.from_dict(state_dict)
                
                # 方案过滤
                if (self.interested_schemas and 
                    current_state.schema_name not in self.interested_schemas):
                    return
                
                # 状态变化检测
                if self.last_state:
                    self._handle_state_change(self.last_state, current_state)
                
                self.last_state = current_state
                
        except (OSError, json.JSONDecodeError):
            pass
    
    def _handle_state_change(self, old_state: RimeState, new_state: RimeState):
        """处理状态变化"""
        # 输入模式切换
        if old_state.input_mode != new_state.input_mode:
            print(f"输入模式切换: {old_state.input_mode} -> {new_state.input_mode}")
        
        # 开始输入
        if not old_state.is_composing and new_state.is_composing:
            print("开始输入")
        
        # 结束输入
        if old_state.is_composing and not new_state.is_composing:
            print("结束输入")
        
        # 输入文本变化
        if old_state.input_text != new_state.input_text:
            print(f"输入变化: '{old_state.input_text}' -> '{new_state.input_text}'")
        
        # 候选词数量变化
        if old_state.candidate_count != new_state.candidate_count:
            print(f"候选词数量: {old_state.candidate_count} -> {new_state.candidate_count}")
    
    def close(self):
        if self.pipe_fd:
            os.close(self.pipe_fd)
            self.pipe_fd = None

# 使用示例
def main():
    processor = RimeStateProcessor()
    
    # 只关注特定的输入方案
    processor.add_schema_filter("wanxiang_pro")
    processor.add_schema_filter("wanxiang_radical")
    
    if not processor.connect():
        print("连接失败")
        return
    
    print("开始处理状态变化...")
    
    try:
        while True:
            processor.process_states()
            time.sleep(0.01)
    except KeyboardInterrupt:
        print("\n停止处理")
    finally:
        processor.close()

if __name__ == "__main__":
    main()
```

### 3. 错误处理和监控

```python
import logging
import time
from typing import Dict

class RimeStateMonitor:
    def __init__(self, pipe_path: str = "/tmp/rime_state_pipe"):
        self.pipe_path = pipe_path
        self.pipe_fd = None
        self.stats = {
            'messages_received': 0,
            'errors': 0,
            'last_message_time': 0
        }
        
        # 设置日志
        logging.basicConfig(level=logging.INFO)
        self.logger = logging.getLogger(__name__)
    
    def get_stats(self) -> Dict:
        """获取监控统计"""
        return self.stats.copy()
    
    def monitor_health(self):
        """监控系统健康状态"""
        current_time = time.time()
        
        # 检查是否长时间没有收到消息
        if (self.stats['last_message_time'] > 0 and 
            current_time - self.stats['last_message_time'] > 5):
            self.logger.warning("长时间未收到消息，可能存在问题")
        
        # 检查错误率
        if self.stats['messages_received'] > 0:
            error_rate = self.stats['errors'] / self.stats['messages_received']
            if error_rate > 0.1:  # 错误率超过10%
                self.logger.warning(f"错误率过高: {error_rate:.2%}")
```
