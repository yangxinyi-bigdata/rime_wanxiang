```
   def _send_to_rime(self, mapped_action: str):
        """将映射动作通过 Rime 配置命令通路下发给客户端。

        使用 rime_socket_serve 的 单客户端配置队列 接口，构造一条通用命令：
        {"command": "hotkey_action", "action": <mapped_action>}。
        由 Rime 端按需解析执行。
        """
        try:
            if not self._rime_server:
                logger.warning(f"↩️ 无 rime_server，无法下发热键动作: {mapped_action}")
                return

            cmd = {
                "command": "hotkey_action",
                "action": str(mapped_action),
                "source": "global_hotkeys",
                "timestamp": int(time.time() * 1000),
            }

            sent_clients = self._rime_server.添加配置变更到所有客户端(cmd)
            if sent_clients:
                logger.info(
                    f"➡️ 已下发热键动作到 Rime 队列: {mapped_action} -> {sent_clients}"
                )
            else:
                logger.info(
                    f"➡️ 已加入待处理队列（当前无就绪客户端）: {mapped_action}"
                )
        except Exception as e:
            logger.error(f"下发热键动作失败: {mapped_action}, 错误: {e}")

```
