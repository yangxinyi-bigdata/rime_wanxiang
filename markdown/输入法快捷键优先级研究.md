## 专题研究： Rime 输入法的快捷键优先级会被其他软件占据

### 问题描述

在 Rime 输入法当中，我设置的快捷键，优先级比其他软件更低，导致我按下快捷键，即使在输入状态下，也会变成触发其他软件的快捷键，而不是 Rime 的功能。

**具体现象：**
- 即使在输入状态（is_composing = true）下，快捷键也被其他软件拦截
- 与快捷键工具（如 Karabiner-Elements、BetterTouchTool 等）冲突
- Rime 中设置的快捷键经常不生效，无法正常使用

### 根本原因分析

**系统限制：** 目前调查下来，基本上没办法让 Rime 输入法获得比其他软件更高优先级的快捷键，因为：
1. Rime 是一个输入法，在 macOS 系统架构中属于 Input Method 层级
2. 系统全局快捷键工具通常运行在更高的权限级别
3. 输入法的事件处理优先级天然低于系统级快捷键工具

### 解决方案总览

基于上述分析，我们需要开发一个外部辅助工具来解决这个问题。核心思路是：**通过第三方软件拦截全局快捷键，然后根据 Rime 的状态智能转发**。

## 方案一：Python 管理助手 + Lua 通信方案

[全局快捷键] → [Python 管理助手] → [判断 Rime 状态] → [转发到 Rime 或其他软件]

**2.3 智能转发模块**

local socket = require("socket")

local function start_tcp_server()
    local server = socket.bind("localhost", 8888)
    server:settimeout(0)  -- 非阻塞模式
    
    while true do
        local client = server:accept()
        if client then
            client:settimeout(0)
            -- 处理客户端请求
            local data = client:receive()
            if data == "GET_STATUS" then

            client:close()
        end
        -- 避免 CPU 占用过高

    end
end

## 快捷键透传技术方案详解

### 问题核心：如何让 Python 拦截快捷键后再"透传"给原应用

当 Python 程序拦截到快捷键后，如果 Rime 不在输入状态，我们需要将快捷键转发给原本应该接收它的应用程序。但是我们并不知道原应用是什么，这就需要实现"透传"机制。

### 方案一：低级别快捷键拦截 + 有条件透传

**核心思路：** 在足够低的系统层级拦截快捷键，如果不需要处理就不拦截，让系统继续分发。

#### 1.1 使用 `pynput` 的条件拦截

```python
from pynput import keyboard
import time
import threading

class ConditionalKeyInterceptor:
    def __init__(self, rime_status_manager):
        self.rime_status_manager = rime_status_manager
        self.current_keys = set()
        self.target_combinations = {
            # 需要拦截的快捷键组合
            frozenset([keyboard.Key.cmd, keyboard.KeyCode.from_char('s')]),
            frozenset([keyboard.Key.tab]),
            frozenset([keyboard.Key.cmd, keyboard.Key.enter]),
        }
        
    def on_press(self, key):
        """按键按下事件"""
        self.current_keys.add(key)
        
        # 检查是否匹配我们关心的快捷键
        current_combo = frozenset(self.current_keys)
        
        for target_combo in self.target_combinations:
            if target_combo.issubset(current_combo):
                # 匹配到目标快捷键，检查是否需要拦截
                if self.should_intercept(target_combo):
                    # 需要拦截，处理快捷键
                    self.handle_intercepted_key(target_combo)
                    return False  # 阻止事件继续传播
                else:
                    # 不需要拦截，让事件继续传播
                    return True
        
        return True  # 默认不拦截
    
    def on_release(self, key):
        """按键释放事件"""
        self.current_keys.discard(key)
        return True
    
    def should_intercept(self, key_combo):
        """判断是否需要拦截这个快捷键"""
        # 检查 Rime 状态
        if not self.rime_status_manager.is_status_fresh():
            return False
        
        return self.rime_status_manager.is_composing
    
    def handle_intercepted_key(self, key_combo):
        """处理被拦截的快捷键"""
        print(f"拦截并处理快捷键: {key_combo}")
        # 这里发送给 Rime 或执行相关操作
        self.send_to_rime(key_combo)
    
    def send_to_rime(self, key_combo):
        """发送快捷键给 Rime"""
        # 实现发送逻辑
        pass
    
    def start_listener(self):
        """启动监听器"""
        with keyboard.Listener(
            on_press=self.on_press,
            on_release=self.on_release
        ) as listener:
            listener.join()
```

#### 1.2 关键点解析

1. **返回值控制透传**：
   - `return False`：阻止事件继续传播（拦截）
   - `return True`：允许事件继续传播（透传）

2. **精确匹配**：只拦截我们关心的快捷键组合

3. **状态判断**：根据 Rime 状态决定是否拦截

import keyboard
import time

class SmartKeyInterceptor:
    def __init__(self, rime_status_manager):
        self.rime_status_manager = rime_status_manager
        self.registered_hotkeys = []
        
    def register_conditional_hotkey(self, hotkey_str, callback):
        """注册条件性快捷键"""
        def conditional_callback():
            if self.should_handle_hotkey():
                callback()
            else:
                # 不处理，让快捷键继续传播
                self.passthrough_hotkey(hotkey_str)
        
        # 注册快捷键但使用条件回调
        keyboard.add_hotkey(hotkey_str, conditional_callback, suppress=True)
        self.registered_hotkeys.append(hotkey_str)
    
    def should_handle_hotkey(self):
        """判断是否应该处理快捷键"""
        return (self.rime_status_manager.is_composing and 
                self.rime_status_manager.is_status_fresh())
    
    def passthrough_hotkey(self, hotkey_str):
        """透传快捷键到系统"""
        # 临时取消注册
        keyboard.remove_hotkey(hotkey_str)
        
        # 重新发送按键序列
        self.simulate_keypress(hotkey_str)
        
        # 延迟后重新注册
        threading.Timer(0.1, lambda: self.re_register_hotkey(hotkey_str)).start()
    
    def simulate_keypress(self, hotkey_str):
        """模拟按键"""
        # 解析快捷键字符串并模拟按键
        keys = hotkey_str.split('+')
        
        # 按下所有修饰键
        for key in keys[:-1]:
            keyboard.press(key.strip())
        
        # 按下并释放主键
        main_key = keys[-1].strip()
        keyboard.press_and_release(main_key)
        
        # 释放所有修饰键
        for key in reversed(keys[:-1]):
            keyboard.release(key.strip())
    
    def re_register_hotkey(self, hotkey_str):
        """重新注册快捷键"""
        callback = self.hotkey_callbacks.get(hotkey_str)
        if callback:
            keyboard.add_hotkey(hotkey_str, callback, suppress=True)

    def __init__(self, rime_status_manager):

            'cmd+s': self.handle_cmd_s,

        self.running = True
        self.monitor_thread = threading.Thread(target=self._monitor_loop)
        self.monitor_thread.daemon = True

        while self.running:
            should_register = (self.rime_status_manager.is_composing and

            if should_register and not self.registered:
                self._register_hotkeys()

                self._unregister_hotkeys()

        """注册快捷键"""

# 文件: rime_http_bridge.py
import json
import os
import threading
import time
from flask import Flask, jsonify
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class RimeStatusBridge:
    def __init__(self):
        self.status = {
            "is_composing": False,
            "current_schema": "",
            "timestamp": 0,
            "input_text": "",
            "candidate_count": 0
        }
        self.pipe_path = "/tmp/rime_status_pipe"
        self.file_path = "/tmp/rime_status.json"
        self.signal_path = "/tmp/rime_status.json.signal"
        
        # 启动数据接收
        self.start_pipe_listener()
        self.start_file_watcher()
    
    def start_pipe_listener(self):
        """启动管道监听器"""
        def listen_pipe():
            while True:
                try:
                    if os.path.exists(self.pipe_path):
                        with open(self.pipe_path, "r") as pipe:
                            for line in pipe:
                                if line.strip():
                                    self.status = json.loads(line.strip())
                                    print(f"管道更新: {self.status}")
                except Exception as e:
                    print(f"管道监听错误: {e}")
                    time.sleep(0.1)
        
        thread = threading.Thread(target=listen_pipe, daemon=True)
        thread.start()
    
    def start_file_watcher(self):
        """启动文件监听器"""
        class StatusFileHandler(FileSystemEventHandler):
            def __init__(self, bridge):
                self.bridge = bridge
            
            def on_created(self, event):
                if event.src_path == self.bridge.signal_path:
                    self.bridge.load_from_file()
            
            def on_modified(self, event):
                if event.src_path == self.bridge.signal_path:
                    self.bridge.load_from_file()
        
        observer = Observer()
        observer.schedule(StatusFileHandler(self), "/tmp/", recursive=False)
        observer.start()
    
    def load_from_file(self):
        """从文件加载状态"""
        try:
            with open(self.file_path, "r") as f:
                self.status = json.load(f)
                print(f"文件更新: {self.status}")
        except Exception as e:
            print(f"文件加载错误: {e}")
    
    def get_status(self):
        """获取当前状态"""
        return self.status.copy()
    
    def is_composing(self):
        """判断是否在输入状态"""
        return self.status.get("is_composing", False)
    
    def is_status_fresh(self, max_age=5):
        """判断状态是否新鲜"""
        current_time = time.time()
        last_update = self.status.get("timestamp", 0)
        return (current_time - last_update) < max_age

from typing import Dict, Callable

class RimeHotkeyManager:
    def __init__(self, bridge_url="http://127.0.0.1:8080"):
        self.bridge_url = bridge_url
        self.hotkey_handlers = {}
        self.last_status_check = 0

    def register_smart_hotkey(self, hotkey_str: str, rime_handler: Callable):
        """注册智能快捷键"""
        def smart_handler():
            if self.should_handle_in_rime():
                print(f"Rime 处理: {hotkey_str}")
                rime_handler()
            else:
                print(f"系统处理: {hotkey_str}")
                self.send_to_system(hotkey_str)
        
        keyboard.add_hotkey(hotkey_str, smart_handler, suppress=True)
        self.hotkey_handlers[hotkey_str] = smart_handler
    
    def should_handle_in_rime(self):
        """判断是否应该在 Rime 处理"""
        # 缓存状态检查以减少 HTTP 请求
        current_time = time.time()
        if current_time - self.last_status_check > 0.1:  # 100ms 缓存
            try:
                response = requests.get(
                    f"{self.bridge_url}/api/is_composing",
                    timeout=0.05  # 50ms 超时
                )
                if response.status_code == 200:
                    self.cached_status = response.json()
                    self.last_status_check = current_time
            except Exception as e:
                print(f"状态检查失败: {e}")
                # 失败时假设不在输入状态
                self.cached_status = {"is_composing": False, "is_fresh": False}
        
        return (self.cached_status.get("is_composing", False) and 
                self.cached_status.get("is_fresh", False))

            set frontApp to name of first application process whose frontmost is true
            tell process frontApp
                {self._generate_keystroke_command(hotkey_str)}
            end tell
        end tell
        '''

package main

import (
    "encoding/json"
    "fmt"
    "io/ioutil"
    "net/http"
    "os"
    "path/filepath"
    "sync"
    "time"
    
    "github.com/fsnotify/fsnotify"
    "github.com/gin-gonic/gin"
)

type RimeStatus struct {
    IsComposing    bool   `json:"is_composing"`
    CurrentSchema  string `json:"current_schema"`
    Timestamp      int64  `json:"timestamp"`
    InputText      string `json:"input_text"`
    CandidateCount int    `json:"candidate_count"`
}

type RimeBridge struct {
    status     RimeStatus
    statusLock sync.RWMutex
    pipePath   string
    filePath   string
    signalPath string
}

func NewRimeBridge() *RimeBridge {
    return &RimeBridge{
        status: RimeStatus{
            IsComposing:    false,
            CurrentSchema:  "",
            Timestamp:      0,
            InputText:      "",
            CandidateCount: 0,
        },
        pipePath:   "/tmp/rime_status_pipe",
        filePath:   "/tmp/rime_status.json",
        signalPath: "/tmp/rime_status.json.signal",
    }
}

func (rb *RimeBridge) startFileWatcher() {
    watcher, err := fsnotify.NewWatcher()
    if err != nil {
        fmt.Printf("创建文件监听器失败: %v\n", err)
        return
    }
    defer watcher.Close()

    // 监听临时目录
    err = watcher.Add(filepath.Dir(rb.signalPath))
    if err != nil {
        fmt.Printf("添加监听路径失败: %v\n", err)
        return
    }

    for {
        select {
        case event, ok := <-watcher.Events:
            if !ok {
                return
            }
            if event.Name == rb.signalPath && event.Op&fsnotify.Create == fsnotify.Create {
                rb.loadFromFile()
            }
        case err, ok := <-watcher.Errors:
            if !ok {
                return
            }
            fmt.Printf("文件监听错误: %v\n", err)
        }
    }
}

func (rb *RimeBridge) loadFromFile() {
    data, err := ioutil.ReadFile(rb.filePath)
    if err != nil {
        fmt.Printf("读取状态文件失败: %v\n", err)
        return
    }

    var status RimeStatus
    if err := json.Unmarshal(data, &status); err != nil {
        fmt.Printf("解析状态文件失败: %v\n", err)
        return
    }

    rb.statusLock.Lock()
    rb.status = status
    rb.statusLock.Unlock()
    
    fmt.Printf("状态更新: %+v\n", status)
}

func (rb *RimeBridge) GetStatus() RimeStatus {
    rb.statusLock.RLock()
    defer rb.statusLock.RUnlock()
    return rb.status
}

func (rb *RimeBridge) IsComposing() bool {
    status := rb.GetStatus()
    return status.IsComposing && rb.IsStatusFresh(5)
}

func (rb *RimeBridge) IsStatusFresh(maxAge int64) bool {
    status := rb.GetStatus()
    return time.Now().Unix()-status.Timestamp < maxAge
}

func main() {
    bridge := NewRimeBridge()
    
    // 启动文件监听
    go bridge.startFileWatcher()
    
    // 创建 HTTP 服务
    gin.SetMode(gin.ReleaseMode)
    r := gin.Default()
    
    r.GET("/api/status", func(c *gin.Context) {
        c.JSON(200, bridge.GetStatus())
    })
    
    r.GET("/api/is_composing", func(c *gin.Context) {
        c.JSON(200, gin.H{
            "is_composing": bridge.IsComposing(),
            "is_fresh":     bridge.IsStatusFresh(5),
        })
    })
    
    r.GET("/api/health", func(c *gin.Context) {
        status := bridge.GetStatus()
        c.JSON(200, gin.H{
            "status":      "ok",
            "last_update": status.Timestamp,
            "is_fresh":    bridge.IsStatusFresh(5),
        })
    })
